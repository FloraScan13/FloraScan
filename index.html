-- Run this SQL script in your Supabase SQL Editor:

create table if not exists plants (
  id text primary key,
  user_id uuid references auth.users(id) on delete cascade,
  name text not null,
  variety text,
  icon_key text,
  hex text,
  date text,
  medium text,
  notes text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

create table if not exists plant_logs (
  id text primary key,
  plant_id text references plants(id) on delete cascade,
  user_id uuid references auth.users(id) on delete cascade,
  date text,
  type text,
  text text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

alter table plants enable row level security;
alter table plant_logs enable row level security;

create policy "Users can manage own plants" on plants for all using (auth.uid() = user_id);
create policy "Users can manage own plant logs" on plant_logs for all using (auth.uid() = user_id);
