async function loadSettingsFromSupabase() { if (!window.supabase) return; try { const { data, error } = await window.supabase.from('settings').select('*').eq('id', 'default').maybeSingle(); if (error) { console.warn('Could not load remote settings:', error.message || error); return; } if (!data) return; if (data.username) settings.username = data.username; if (data.logo_url) settings.logoUrl = data.logo_url; setStored('flora_settings', JSON.stringify(settings)); applyBranding(); } catch (err) { console.warn('Remote settings unavailable:', err); } } async function saveSettingsToSupabase() { if (!window.supabase) return; try { const payload = { id: 'default', username: settings.username, logo_url: settings.logoUrl, supabase_url: document.getElementById('setting-supabase-url')?.value || getStored('flora_supabase_url') || '', supabase_key: document.getElementById('setting-supabase-key')?.value || getStored('flora_supabase_key') || '' }; const { error } = await window.supabase.from('settings').upsert(payload, { onConflict: 'id' }); if (error) { console.warn('Remote settings save failed:', error.message || error); } else { console.info('Remote settings saved to Supabase'); } } catch (err) { console.warn('Failed to sync settings to Supabase:', err); } }
